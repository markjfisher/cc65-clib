memory load atari "${BINARY_PATH}"
symbols load "${SYMBOLS_PATH}"

; File mock: contents "ABCDE", length 5, handle $11.
memory write #0x0900 0x(41,42,43,44,45)
memory write #0x0A00 0x(05)
memory write #0x0A03 0x(11)

registers flush

run $init_harness until (CP = $end_init_harness OR cycle_count > 800000)
assert CP = $end_init_harness $$completed correctly$$

assert $_test_result = 0x01      $$open() succeeded$$
assert $_test_result + 1 = 0x02  $$lseek(SEEK_SET,2) returned 2$$
assert $_test_result + 2 = 0x03  $$lseek(SEEK_CUR,+1) used pending seek$$
assert $_test_result + 3 = 0x44  $$read after pending seek = 'D'$$
assert $_test_result + 4 = 0x04  $$mock file pointer advanced to 4$$
assert $_test_result + 5 = 0x03  $$lseek(SEEK_CUR,-1) used actual file ptr$$
assert $_test_result + 6 = 0x44  $$read after backstep = 'D'$$
assert $_test_result + 7 = 0x04  $$lseek(SEEK_END,-1) returned 4$$
assert $_test_result + 8 = 0xFF  $$lseek past BOF returned -1$$
assert $_test_result + 9 = 0x07  $$errno = EINVAL$$
assert $_test_result + 10 = 0x00 $$lseek(SEEK_SET,0) returned 0$$
assert $_test_result + 11 = 0x41 $$read after rewind = 'A'$$
assert $_test_result + 12 = 0x01 $$completed$$

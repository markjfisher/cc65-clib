memory load atari "${BINARY_PATH}"
symbols load "${SYMBOLS_PATH}"
memory write $mock_print_armed 0x(01)

; Mock VDU workspace with cursor at (10,5) — VDU_WKSP=$300 + VDUVAR_TEXT_CURSOR_XY=$18
; X at $318, Y at $319
memory write #0x0318 0x(0A,05)

registers flush

run $init_harness until (CP = $end_init_harness OR cycle_count > 1000)
assert CP = $end_init_harness $$completed correctly$$

; Initial cursor position from VDU mock
assert $_test_result = 0x0A     $$wherex = 10$$
assert $_test_result + 1 = 0x05 $$wherey = 5$$

; gotoxy(10,5) sends VDU 31, x, y via OSWRCH (x on stack, y in A)
; VDU 31 = 0x1F, X=10 = 0x0A, Y=5 = 0x05
assert #0xC800 = 0x1F     $$VDU 31 sent$$
assert #0xC801 = 0x0A     $$X=10 sent$$
assert #0xC802 = 0x05     $$Y=5 sent$$

; VDU workspace not updated by harness — wherex/y still read mock values
assert $_test_result + 2 = 0x0A $$wherex still 10 (VDU workspace unchanged)$$
assert $_test_result + 3 = 0x05 $$wherey still 5$$

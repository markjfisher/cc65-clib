memory load atari "${BINARY_PATH}"
symbols load "${SYMBOLS_PATH}"
memory write $mock_print_armed 0x(01)

registers flush

run $init_harness until (CP = $end_init_harness OR cycle_count > 500)
assert CP = $end_init_harness $$completed correctly$$

; gotox(10) sends VDU 31, X via OSWRCH
assert #0xC800 = 0x1F     $$VDU 31 sent$$
assert #0xC801 = 0x0A     $$X=10 sent$$

; gotoy(5) sends VDU 31, current X, Y via OSWRCH
; Current X read from VDU workspace ($318) — in harness this is 0
assert #0xC802 = 0x00     $$current X from VDU workspace = 0$$
assert #0xC803 = 0x1F     $$VDU 31 for gotoy$$
assert #0xC804 = 0x00     $$current X from VDU workspace = 0$$
assert #0xC805 = 0x05     $$Y=5 sent$$

assert $_test_result = 0x01     $$gotox/gotoy completed$$

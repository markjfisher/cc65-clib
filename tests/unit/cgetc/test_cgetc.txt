marker $$ cgetc test $$

memory load atari "${BINARY_PATH}"
symbols load "${SYMBOLS_PATH}"

; Set up mock character buffer for OSRDCH — provides 'X', Escape, 'Y'
memory write #0x1E00 0x(58,1B,59,00)

registers flush

run $init_harness until (CP = $end_init_harness OR cycle_count > 5000)
assert CP = $end_init_harness $$completed correctly$$

assert $_test_result = 0x58     $$first cgetc returned 'X' low byte (0x58)$$
assert $_test_result + 1 = 0x00 $$first cgetc returned zero high byte$$
assert $_test_result + 2 = 0x1B $$second cgetc returned Escape low byte (0x1B)$$
assert $_test_result + 3 = 0x00 $$second cgetc returned zero high byte$$
assert $_test_result + 4 = 0x59 $$third cgetc returned 'Y' low byte (0x59)$$
assert $_test_result + 5 = 0x00 $$third cgetc returned zero high byte$$
